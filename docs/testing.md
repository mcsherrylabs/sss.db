## Test Setup

### In-Memory Database for Tests

Use HSQLDB for fast in-memory tests:

**build.sbt:**
```scala
libraryDependencies += "org.hsqldb" % "hsqldb" % "2.6.1" % Test
libraryDependencies += "org.scalatest" %% "scalatest" % "3.2.2" % Test
```

**src/test/resources/application.conf:**
```hocon
testDb {
  datasource {
    driver = "org.hsqldb.jdbc.JDBCDriver"
    connection = "jdbc:hsqldb:mem:test"  # In-memory
    user = "SA"
    pass = ""
    maxPoolSize = 5
  }

  viewCachesSize = 100
  useShutdownHook = false  # Tests manage lifecycle

  createSql = [
    """CREATE TABLE IF NOT EXISTS users (
         id BIGINT GENERATED BY DEFAULT AS IDENTITY,
         name VARCHAR(256),
         email VARCHAR(256),
         age INTEGER,
         active BOOLEAN,
         PRIMARY KEY(id)
       )"""
  ]
}
```

## Test Fixtures

### Basic Test Setup

```scala
import org.scalatest.flatspec.AnyFlatSpec
import org.scalatest.matchers.should.Matchers
import org.scalatest.{BeforeAndAfterEach, BeforeAndAfterAll}
import sss.db._
import scala.concurrent.ExecutionContext.Implicits.global

class UserServiceSpec extends AnyFlatSpec
  with Matchers
  with BeforeAndAfterEach
  with BeforeAndAfterAll {

  // Create context and database
  implicit val sync: SyncRunContext = new SyncRunContext(global)
  var db: Db = _
  var userTable: Table = _

  override def beforeAll(): Unit = {
    db = Db("testDb")
    userTable = db.table("users")
  }

  override def beforeEach(): Unit = {
    // Clear table before each test
    userTable.delete(where("")).runSyncAndGet
  }

  override def afterAll(): Unit = {
    db.shutdown.runSyncAndGet
  }

  "UserService" should "create user" in {
    val user = userTable.persist(Map(
      "name" -> "Alice",
      "email" -> "alice@example.com"
    )).runSyncAndGet

    user.string("name") shouldBe "Alice"
    user.string("email") shouldBe "alice@example.com"
  }
}
```

### Reusable Test Fixture

```scala
trait DbTestFixture extends BeforeAndAfterEach with BeforeAndAfterAll {
  self: Suite =>

  implicit val sync: SyncRunContext = new SyncRunContext(
    ExecutionContext.Implicits.global
  )

  var db: Db = _
  var userTable: Table = _

  override def beforeAll(): Unit = {
    db = Db("testDb")
    userTable = db.table("users")
  }

  override def beforeEach(): Unit = {
    // Clean slate for each test
    userTable.delete(where("")).runSyncAndGet
  }

  override def afterAll(): Unit = {
    db.shutdown.runSyncAndGet
  }
}

// Use in tests
class UserServiceSpec extends AnyFlatSpec with Matchers with DbTestFixture {

  "UserService" should "find user by email" in {
    userTable.persist(Map("name" -> "Alice", "email" -> "alice@example.com"))
      .runSyncAndGet

    val found = userTable.find(where(ps"email = ${"alice@example.com"}"))
      .runSyncAndGet

    found shouldBe defined
    found.get.string("name") shouldBe "Alice"
  }
}
```

## Testing FutureTx Operations

### Testing with SyncRunContext

Simplest approach for tests:

```scala
"persist" should "insert new user" in {
  val operation = userTable.persist(Map("name" -> "Bob"))

  // Execute and get result
  val user = operation.runSyncAndGet

  user.string("name") shouldBe "Bob"
  user.id should be > 0L
}
```

### Testing with Try Pattern

```scala
"persist" should "handle constraint violations" in {
  // Create user
  userTable.persist(Map("email" -> "unique@example.com")).runSyncAndGet

  // Try to create duplicate (assume unique constraint on email)
  val result = userTable.persist(Map("email" -> "unique@example.com")).runSync

  result.isFailure shouldBe true
  result.failed.get shouldBe a[DbException]
}
```

### Testing Transactions

```scala
"transaction" should "rollback on failure" in {
  val transaction = for {
    user <- userTable.persist(Map("name" -> "Charlie"))
    _ <- userTable.persist(Map("invalid" -> "data"))  // Will fail
  } yield user

  val result = transaction.runSync

  result.isFailure shouldBe true

  // Verify nothing was inserted (transaction rolled back)
  val count = userTable.count().runSyncAndGet
  count shouldBe 0
}
```

### Testing Async Operations

```scala
import org.scalatest.concurrent.ScalaFutures
import scala.concurrent.duration._

class AsyncUserServiceSpec extends AsyncFlatSpec
  with Matchers
  with ScalaFutures {

  implicit val async: AsyncRunContext = new AsyncRunContext(
    ExecutionContext.Implicits.global
  )

  "persist" should "insert user asynchronously" in {
    val db = Db("testDb")
    val userTable = db.table("users")

    val futureUser = userTable.persist(Map("name" -> "Dave")).run

    futureUser.map { user =>
      user.string("name") shouldBe "Dave"
    }
  }
}
```

## Test Database Lifecycle

### Creating Tables in Tests

```scala
class UserServiceSpec extends AnyFlatSpec with Matchers {

  val db = Db("testDb")

  override def beforeAll(): Unit = {
    // Create table if not exists
    db.executeSql("""
      CREATE TABLE IF NOT EXISTS test_users (
        id BIGINT GENERATED BY DEFAULT AS IDENTITY,
        name VARCHAR(256),
        PRIMARY KEY(id)
      )
    """).runSyncAndGet
  }

  override def afterAll(): Unit = {
    // Drop table after tests
    db.executeSql("DROP TABLE IF EXISTS test_users").runSyncAndGet
    db.shutdown.runSyncAndGet
  }
}
```

### Seeding Test Data

```scala
trait SeedData {
  def seedUsers(table: Table)(implicit sync: SyncRunContext): Seq[Row] = {
    val users = List(
      Map("name" -> "Alice", "age" -> 30),
      Map("name" -> "Bob", "age" -> 25),
      Map("name" -> "Charlie", "age" -> 35)
    )

    val inserts = users.map(table.insert)
    FutureTx.sequence(inserts).runSyncAndGet
  }
}

class UserQuerySpec extends AnyFlatSpec with Matchers with DbTestFixture with SeedData {

  "filter" should "find users by age" in {
    seedUsers(userTable)

    val youngUsers = userTable.filter(
      where(ps"age < ${30}")
    ).runSyncAndGet

    youngUsers.length shouldBe 1
    youngUsers.head.string("name") shouldBe "Bob"
  }
}
```

## Example Test Patterns

### Testing CRUD Operations

```scala
class UserCrudSpec extends AnyFlatSpec with Matchers with DbTestFixture {

  "create" should "insert new user" in {
    val user = userTable.persist(Map("name" -> "Eve")).runSyncAndGet
    user.id should be > 0L
  }

  "read" should "find user by id" in {
    val created = userTable.persist(Map("name" -> "Frank")).runSyncAndGet
    val found = userTable(created.id).runSyncAndGet

    found.string("name") shouldBe "Frank"
  }

  "update" should "modify user" in {
    val user = userTable.persist(Map("name" -> "Grace")).runSyncAndGet

    userTable.update(
      Map("name" -> "Grace Smith"),
      where(ps"id = ${user.id}")
    ).runSyncAndGet

    val updated = userTable(user.id).runSyncAndGet
    updated.string("name") shouldBe "Grace Smith"
  }

  "delete" should "remove user" in {
    val user = userTable.persist(Map("name" -> "Henry")).runSyncAndGet

    userTable.delete(where(ps"id = ${user.id}")).runSyncAndGet

    val found = userTable.find(where(ps"id = ${user.id}")).runSyncAndGet
    found shouldBe None
  }
}
```

### Testing Query Filters

```scala
"filter" should "find active users" in {
  // Seed data
  userTable.persist(Map("name" -> "Active1", "active" -> true)).runSyncAndGet
  userTable.persist(Map("name" -> "Inactive", "active" -> false)).runSyncAndGet
  userTable.persist(Map("name" -> "Active2", "active" -> true)).runSyncAndGet

  // Test filter
  val active = userTable.filter(where(ps"active = ${true}")).runSyncAndGet

  active.length shouldBe 2
  active.map(_.string("name")) should contain allOf ("Active1", "Active2")
}
```

### Testing Error Handling

```scala
"persist" should "throw on constraint violation" in {
  userTable.persist(Map("email" -> "test@example.com")).runSyncAndGet

  an[DbException] should be thrownBy {
    userTable.persist(Map("email" -> "test@example.com")).runSyncAndGet
  }
}
```

### Testing Optimistic Locking

```scala
"persist" should "handle version conflicts" in {
  // Create versioned row
  val account = accountTable.persist(Map(
    "balance" -> 100.0,
    "version" -> 0L
  )).runSyncAndGet

  // Simulate concurrent update
  accountTable.persist(Map(
    "id" -> account.id,
    "balance" -> 150.0,
    "version" -> account.long("version")
  )).runSyncAndGet  // Increments version

  // Try to update with stale version
  val result = accountTable.persist(Map(
    "id" -> account.id,
    "balance" -> 200.0,
    "version" -> account.long("version")  // Stale version!
  )).runSync

  result.isFailure shouldBe true
  result.failed.get shouldBe a[DbOptimisticLockingException]
}
```

## Summary

**Test Setup:**
- Use HSQLDB in-memory for fast tests
- Configure in test resources
- Use SyncRunContext for simplicity

**Fixtures:**
- Create reusable test traits
- Clean database before each test
- Shutdown cleanly after all tests

**Testing Patterns:**
- Use `.runSyncAndGet` for simple assertions
- Use `.runSync` with pattern matching for error testing
- Test transactions with rollback verification
- Seed test data with helper functions

**Best Practices:**
- Isolate tests (clean state before each)
- Test both success and failure paths
- Use meaningful test data
- Test transactions and error handling
- Keep tests fast (in-memory database)
